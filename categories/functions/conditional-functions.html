条件函数 ​

条件函数用于根据条件返回不同的值。

常用条件函数 ​

函数说明示例
IF()条件判断IF(条件, 值1, 值2)
IFNULL()NULL 值替换IFNULL(NULL, '默认值')
NULLIF()NULL 值判断NULLIF(a, b)
CASE条件表达式CASE WHEN ... THEN ... END

函数详解 ​

IF() - 条件判断 ​

sql
-- 语法: IF(condition, value_if_true, value_if_false)
SELECT IF(10 > 5, '大于', '小于');
-- 结果: '大于'

-- 实际应用
SELECT 
    name,
    salary,
    IF(salary > 8000, '高薪资', '普通薪资') AS salary_level
FROM employees;

IFNULL() - NULL 值替换 ​

sql
-- 语法: IFNULL(expression, replacement)
SELECT IFNULL(NULL, '默认值');
-- 结果: '默认值'

SELECT IFNULL(name, '未知') FROM employees;
-- 如果 name 为 NULL，返回 '未知'

-- 实际应用：计算总奖金
SELECT 
    name,
    salary + IFNULL(bonus, 0) AS total_income
FROM employees;

NULLIF() - NULL 值判断 ​

sql
-- 语法: NULLIF(value1, value2)
-- 如果 value1 = value2，返回 NULL；否则返回 value1
SELECT NULLIF(5, 5);    -- NULL
SELECT NULLIF(5, 10);    -- 5

-- 实际应用：避免除零错误
SELECT 
    salary / NULLIF(commission, 0)
FROM employees;
-- 当 commission 为 0 时，不会报错，返回 NULL

CASE - 条件表达式 ​

简单 CASE ​

sql
-- 语法1: CASE 表达式 WHEN 值 THEN 结果 ... END
SELECT 
    department,
    CASE department 
        WHEN 'IT' THEN '技术部'
        WHEN 'HR' THEN '人力资源'
        WHEN 'Sales' THEN '销售部'
        ELSE '其他部门'
    END AS dept_name
FROM employees;

COALESCE() - 返回第一个非 NULL 值 ​

sql
-- 语法: COALESCE(value1, value2, ...)
SELECT COALESCE(NULL, NULL, '第一个值', '第二个值');
-- 结果: '第一个值'

-- 实际应用
SELECT 
    name,
    COALESCE(phone_mobile, phone_home, phone_office, '无电话') AS contact_phone
FROM employees;

综合示例 ​

sql
SELECT 
    name,
    department,
    salary,
    CASE 
        WHEN salary >= 10000 AND department = 'IT' THEN '高级技术'
        WHEN salary >= 8000 AND department = 'IT' THEN '中级技术'
        WHEN department = 'Sales' AND salary >= 7000 THEN '销售精英'
        ELSE '普通员工'
    END AS employee_level,
    IFNULL(bonus, 0) + salary AS total_income
FROM employees
ORDER BY total_income DESC;
