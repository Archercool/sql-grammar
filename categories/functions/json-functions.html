JSON 函数 ​

JSON 函数用于处理 JSON 类型的数据库字段。

常用 JSON 函数 ​

函数说明示例
JSON_EXTRACT()提取 JSON 值JSON_EXTRACT(json, '$.key')
JSON_SET()设置 JSON 值JSON_SET(json, '$.key', value)
JSON_INSERT()插入 JSON 值JSON_INSERT(json, '$.key', value)
JSON_REPLACE()替换 JSON 值JSON_REPLACE(json, '$.key', value)
JSON_REMOVE()删除 JSON 值JSON_REMOVE(json, '$.key')
JSON_KEYS()获取所有键JSON_KEYS(json)
JSON_VALUES()获取所有值JSON_VALUES(json)
JSON_CONTAINS()包含判断JSON_CONTAINS(json, value)
JSON_SEARCH()搜索JSON_SEARCH(json, 'one', 'word')

JSON 数据类型 ​

MySQL 5.7+ 支持原生 JSON 数据类型：

sql
CREATE TABLE user_profiles (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    info JSON
);

函数详解 ​

JSON_EXTRACT() - 提取 JSON 值 ​

sql
-- 语法: JSON_EXTRACT(json_column, '$.path')
SELECT JSON_EXTRACT('{"name": "张三", "age": 30}', '$.name');
-- 结果: '"张三"'

SELECT JSON_EXTRACT('{"user": {"name": "张三"}}', '$.user.name');
-- 结果: '"张三"'

SELECT JSON_EXTRACT('[1, 2, 3]', '$[0]');
-- 结果: '1'

SELECT JSON_EXTRACT('{"skills": ["SQL", "Python"]}', '$.skills[0]');
-- 结果: '"SQL"'

sql

-- 结果: 张三

-- 简写形式: column->>'$.path'
SELECT '{"name": "张三"}'->>'$.name';

JSON_SET() - 设置值 ​

sql
-- 语法: JSON_SET(json, '$.path', value)
SELECT JSON_SET('{"name": "张三"}', '$.age', 30);
-- 结果: '{"name": "张三", "age": 30}'

SELECT JSON_SET('{"name": "张三"}', '$.name', '李四', '$.age', 30);
-- 结果: '{"name": "李四", "age": 30}'

JSON_SEARCH() - 搜索 ​

sql
-- 语法: JSON_SEARCH(json, 'one|all', search_string)
SELECT JSON_SEARCH('{"name": "张三", "city": "北京"}', 'one', '北京');
-- 结果: '$.city'

SELECT JSON_SEARCH('{"skills": ["SQL", "Python"]}', 'one', 'SQL');
-- 结果: '$.skills[0]'

-- 使用通配符
SELECT JSON_SEARCH('{"skills": ["SQL", "Python", "Java"]}', 'one', 'P%');
-- 结果: '$.skills[1]'

综合示例 ​

表中 JSON 操作 ​

sql
-- 创建表
CREATE TABLE products (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    attributes JSON
);

-- 插入数据
INSERT INTO products VALUES 
(1, '手机', '{"color": "黑色", "storage": "256GB", "price": 4999}'),
(2, '电脑', '{"color": "银色", "storage": "512GB", "price": 8999}');

-- 查询 JSON 字段
SELECT 
    name,
    attributes->>'$.color' AS color,
    attributes->>'$.storage' AS storage,
    attributes->>'$.price' AS price
FROM products;

-- 搜索 JSON 中的值
SELECT * FROM products 
WHERE JSON_SEARCH(attributes, 'one', '黑色') IS NOT NULL;

-- 更新 JSON
UPDATE products 
SET attributes = JSON_SET(attributes, '$.warranty', '2年')
WHERE id = 1;
